技術採用：React 19 + Tailwind CSS + @media print

以 React 19 開發可具備「AI後臺動態調整」優勢。

// ProjectCard.jsx — React 19 + Tailwind CSS + @media print 排版範例 (未來 Mac 部署架構)
import { use } from 'react';

export function SiteProgressBoard({ statusPromise }) {
  // 1. React 19 `use` Hook：直接讀取安弟 AI 每 5 分鐘覆寫的 status.json (支援 5~7 個動態案件)
  const projects = use(statusPromise);

  return (
    <section className="grid grid-cols-5 lg:grid-cols-7 gap-3 print:w-[297mm] print:gap-2">
      {projects.map((item) => (
        // 2. 高低交錯排版：even:mt-8 100% 還原 Illustrator/PPT 交錯視覺，列印時自動微調間距
        <article key={item.id} className="even:mt-8 print:even:mt-4 bg-white shadow border print:break-inside-avoid">
          
          {/* <案件名>：鎖定高度與單行截斷 truncate */}
          <header style={{ backgroundColor: item.color }} className="h-16 flex items-center justify-center text-white font-bold truncate px-2">
            {item.projectName}
          </header>

          {/* <本日施工項目> + 現場照片：強制 object-cover 自動裁切比例 */}
          <div className="aspect-[16/10] overflow-hidden">
            <img src={item.photoUrl} className="w-full h-full object-cover" alt={item.todayWork} />
          </div>
          <div className="bg-[#1b3b32] text-white text-center py-1.5 text-xs font-bold truncate px-2">
            本日施工：{item.todayWork}
          </div>

          {/* <說明>：透過 line-clamp-2 嚴格控制最大字數與行數，徹底防呆防跑版 */}
          <p className="p-2.5 text-xs text-gray-600 line-clamp-2 min-h-[3rem] bg-gray-50">
            {item.description}
          </p>

          {/* <施工總進度> & <明日預定工程>：亮色框重點對應副董室列管案 */}
          <footer className={item.isHighlight ? "border-2 border-amber-500 border-b-4 border-b-green-700 p-2.5" : "border p-2.5"}>
            <div className="text-xs font-bold text-green-800">總進度：{item.totalProgress}%</div>
            <div className="text-xs text-gray-700 truncate mt-1">明日預定：{item.tomorrowPlan}</div>
          </footer>
        </article>
      ))}
    </section>
  );
}
